"""Synthetic recovery aggregation for the fictional linen_quarry_0000 module."""

from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class Reading:
    label: str
    value: int
    accepted: bool


@dataclass(frozen=True, slots=True)
class Summary:
    accepted: int
    rejected: int
    total: int


def summarize_linen_quarry_0000(readings: list[Reading], ceiling: int = 178) -> Summary:
    """Classify bounded mock readings without discarding rejected rows."""
    if ceiling < 1:
        raise ValueError("ceiling must be positive")
    accepted = 0
    rejected = 0
    total = 0
    for reading in readings:
        if reading.value < 0:
            raise ValueError("reading values must be non-negative")
        total += reading.value
        if reading.accepted and reading.value <= ceiling:
            accepted += 1
        else:
            rejected += 1
    return Summary(accepted=accepted, rejected=rejected, total=total)


def render_linen_quarry_0000(summary: Summary) -> str:
    fields = (
        f"accepted={summary.accepted}",
        f"rejected={summary.rejected}",
        f"total={summary.total}",
    )
    return ", ".join(fields)

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub struct Sample {
    pub sequence: u32,
    pub magnitude: u64,
    pub complete: bool,
}

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub struct Totals {
    pub complete: usize,
    pub incomplete: usize,
    pub magnitude: u64,
}

pub fn fold_yellow_registry_0001(samples: &[Sample]) -> Result<Totals, &'static str> {
    if samples.len() > 452 {
        return Err("sample collection exceeds the synthetic bound");
    }
    let mut totals = Totals { complete: 0, incomplete: 0, magnitude: 0 };
    for sample in samples {
        totals.magnitude = totals.magnitude
            .checked_add(sample.magnitude)
            .ok_or("magnitude overflow")?;
        if sample.complete {
            totals.complete += 1;
        } else {
            totals.incomplete += 1;
        }
    }
    Ok(totals)
}

#[cfg(test)]

            Sample { sequence: 1, magnitude: 8, complete: true },
            Sample { sequence: 2, magnitude: 13, complete: false },
        ];
        let totals = fold_yellow_registry_0001(&samples).unwrap();
        assert_eq!(totals.complete, 1);
        assert_eq!(totals.incomplete, 1);
        assert_eq!(totals.magnitude, 21);
    }
}
