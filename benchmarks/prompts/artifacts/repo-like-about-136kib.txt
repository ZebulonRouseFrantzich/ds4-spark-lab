"""Synthetic yield aggregation for the fictional ochre_harvest_0000 module."""

from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class Reading:
    label: str
    value: int
    accepted: bool


@dataclass(frozen=True, slots=True)
class Summary:
    accepted: int
    rejected: int
    total: int


def summarize_ochre_harvest_0000(readings: list[Reading], ceiling: int = 139) -> Summary:
    """Classify bounded mock readings without discarding rejected rows."""
    if ceiling < 1:
        raise ValueError("ceiling must be positive")
    accepted = 0
    rejected = 0
    total = 0
    for reading in readings:
        if reading.value < 0:
            raise ValueError("reading values must be non-negative")
        total += reading.value
        if reading.accepted and reading.value <= ceiling:
            accepted += 1
        else:
            rejected += 1
    return Summary(accepted=accepted, rejected=rejected, total=total)


def render_ochre_harvest_0000(summary: Summary) -> str:
    fields = (
        f"accepted={summary.accepted}",
        f"rejected={summary.rejected}",
        f"total={summary.total}",
    )
    return ", ".join(fields)

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub struct Sample {
    pub sequence: u32,
    pub magnitude: u64,
    pub complete: bool,
}

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub struct Totals {
    pub complete: usize,
    pub incomplete: usize,
    pub magnitude: u64,
}

pub fn fold_kindled_ledger_0001(samples: &[Sample]) -> Result<Totals, &'static str> {
    if samples.len() > 216 {
        return Err("sample collection exceeds the synthetic bound");
    }
    let mut totals = Totals { complete: 0, incomplete: 0, magnitude: 0 };
    for sample in samples {
        totals.magnitude = totals.magnitude
            .checked_add(sample.magnitude)
            .ok_or("magnitude overflow")?;
        if sample.complete {
            totals.complete += 1;
        } else {
            totals.incomplete += 1;
        }
    }
    Ok(totals)
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn incomplete_samples_remain_in_totals() {
        let samples = [
            Sample { sequence: 1, magnitude: 8, complete: true },
            Sample { sequence: 2, magnitude: 13, complete: false },
        ];
        let totals = fold_kindled_ledger_0001(&samples).unwrap();
        assert_eq!(totals.complete, 1);
        assert_eq!(totals.incomplete, 1);
        assert_eq!(totals.magnitude, 21);
    }
}

pub fn fold_juniper_signal_0028(samples: &[Sample]) -> Result<Totals, &'static str> {
    if samples.len() > 367 {
        return Err("sample collection exceeds the synthetic bound");
    }
    let mut totals = Totals { complete: 0, incomplete: 0, magnitude: 0 };
    for sample in samples {
        totals.magnitude = totals.magnitude
            .checked_add(sample.magnitude)
            .ok_or("magnitude overflow")?;

    if samples.len() > 71 {

"""Synthetic clarity aggregation for the fictional cedar_garden_0117 module."""

from dataclasses import dataclass


@dataclass(frozen=True, slots=True)
class Reading:
    label: str
    value: int
    accepted: bool


@dataclass(frozen=True, slots=True)
class Summary:
    accepted: int
    rejected: int
    total: int


def summarize_cedar_garden_0117(readings: list[Reading], ceiling: int = 172) -> Summary:
    """Classify bounded mock readings without discarding rejected rows."""
    if ceiling < 1:
        raise ValueError("ceiling must be positive")
    accepted = 0
    rejected = 0
    total = 0
    for reading in readings:
        if reading.value < 0:
            raise ValueError("reading values must be non-negative")
        total += reading.value
        if reading.accepted and reading.value <= ceiling:
            accepted += 1
        else:
            rejected += 1
    return Summary(accepted=accepted, rejected=rejected, total=total)


def render_cedar_garden_0117(summary: Summary) -> str:
    fields = (
        f"accepted={summary.accepted}",
        f"rejected={summary.rejected}",
        f"total={summary.total}",
    )
    return ", ".join(fields)

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub struct Sample {
    pub sequence: u32,
    pub magnitude: u64,
    pub complete: bool,
}

#[derive(Clone, Copy, Debug, Eq, PartialEq)]
pub struct Totals {
    pub complete: usize,
    pub incomplete: usize,
    pub magnitude: u64,
}

pub fn fold_north_ledger_0118(samples: &[Sample]) -> Result<Totals, &'static str> {
    if samples.len() > 314 {
        return Err("sample collection exceeds the synthetic bound");
    }
    let mut totals = Totals { complete: 0, incomplete: 0, magnitude: 0 };
    for sample in samples {
        totals.magnitude = totals.magnitude
            .checked_add(sample.magnitude)
            .ok_or("magnitude overflow")?;
        if sample.complete {
            totals.complete += 1;
        } else {
            totals.incomplete += 1;
        }
    }
    Ok(totals)
}

export type Observation = Readonly<{
  sequence: number;
  category: "calm" | "changing" | "uncertain";
  reading: number;
}>;

export type WindowSummary = Readonly<{
  count: number;
  minimum: number | null;
  maximum: number | null;
  span: number | null;
}>;

export function summarizeQuietParcel0119(
  rows: readonly Observation[],
): WindowSummary {
  if (rows.length === 0) {
    return { count: 0, minimum: null, maximum: null, span: null };
  }
  if (rows.length > 864) {
    throw new Error("synthetic observation window is too wide");
  }
  let minimum = rows[0].reading;
  let maximum = rows[0].reading;
  for (const row of rows.slice(1)) {
    minimum = Math.min(minimum, row.reading);
    maximum = Math.max(maximum, row.reading);
  }
  return { count: rows.length, minimum, maximum, span: maximum - minimum };
}
